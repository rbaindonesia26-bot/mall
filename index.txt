                  </div>
                  <div className="px-2.5">
                    <div className="flex items-center gap-1 text-[9px] text-gray-500 mb-2 bg-gray-50 p-1 rounded-lg border border-gray-100">
                      <Store size={10} className="text-emerald-500 shrink-0"/>
                      <span className="truncate flex-1 font-medium">{product.seller}</span>
                      {product.isVerified && <CheckCircle size={10} className="text-blue-500 fill-blue-50 shrink-0"/>}
                    </div>
                    <button onClick={() => addToCart(product)} className="w-full bg-emerald-50 text-emerald-700 border border-emerald-200 py-2 rounded-xl text-xs font-black flex items-center justify-center gap-1 hover:bg-emerald-500 hover:text-white transition">
                      <Plus size={14} /> Tambah
                    </button>
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>
      </div>
    );
  };

  const renderCart = () => (
    <div className="pb-24 bg-gray-50 min-h-screen">
      <div className="bg-white p-4 shadow-sm border-b sticky top-0 z-40 flex items-center justify-center">
        <h1 className="text-base sm:text-lg font-black text-gray-800 flex items-center gap-2"><ShoppingBag className="text-emerald-600"/> Keranjang Belanja</h1>
      </div>
      <div className="p-3 sm:p-4 space-y-4">
        {cart.length === 0 ? (
          <div className="text-center py-20 flex flex-col items-center px-4">
            <ShoppingCart size={50} className="text-gray-200 mb-3" />
            <h3 className="font-bold text-gray-700 mb-1 text-sm">Keranjang Masih Kosong</h3>
            <p className="text-gray-500 text-xs mb-4">Yuk cari barang menarik dan berkah hari ini!</p>
            <button onClick={() => setActiveTab('home')} className="bg-emerald-600 text-white font-bold py-2 px-6 rounded-full text-xs shadow-md">Belanja Sekarang</button>
          </div>
        ) : (
          Object.keys(groupedCart).map((waNumber) => {
            const group = groupedCart[waNumber];
            return (
              <div key={waNumber} className="bg-white rounded-2xl shadow-sm border border-gray-100 overflow-hidden">
                <div className="bg-emerald-50 p-3 border-b border-emerald-100 flex items-center gap-2">
                  <Store size={16} className="text-emerald-600 shrink-0"/>
                  <h3 className="font-bold text-emerald-800 text-xs sm:text-sm truncate">Pesanan dari {group.sellerName}</h3>
                </div>
                <div className="p-3 space-y-3">
                  {group.items.map(item => (
                    <div key={item.id} className="flex gap-2.5 sm:gap-3 items-center">
                      <img src={item.image} alt={item.name} className="w-16 h-16 sm:w-20 sm:h-20 object-cover rounded-xl border border-gray-100 shrink-0" />
                      <div className="flex-1 min-w-0">
                        <h4 className="text-xs sm:text-sm font-medium text-gray-800 line-clamp-2 leading-tight">{item.name}</h4>
                        <p className="text-emerald-600 font-black text-xs sm:text-sm mt-0.5">{formatRupiah(item.price)}</p>
                        <div className="flex items-center justify-between mt-2">
                          <button onClick={() => removeFromCart(item.id)} className="text-gray-400 hover:text-red-500 text-[10px] flex items-center gap-0.5 font-medium"><Trash2 size={12}/> Hapus</button>
                          <div className="flex items-center bg-gray-50 border border-gray-200 rounded-lg p-0.5">
                            <button onClick={() => updateCartQty(item.id, -1)} className="p-1 text-gray-500 hover:text-emerald-600"><Minus size={12}/></button>
                            <span className="w-6 text-center text-xs font-bold">{item.qty}</span>
                            <button onClick={() => updateCartQty(item.id, 1)} className="p-1 text-gray-500 hover:text-emerald-600"><Plus size={12}/></button>
                          </div>
                        </div>
                      </div>
                    </div>
                  ))}
                </div>
                <div className="p-3.5 bg-gray-50 border-t border-gray-100">
                  <div className="flex justify-between items-center mb-3">
                    <span className="text-xs text-gray-600 font-medium">Subtotal</span>
                    <span className="font-black text-base text-gray-800">{formatRupiah(group.total)}</span>
                  </div>
                  <button onClick={() => checkoutToWA(waNumber, group)} className="w-full bg-emerald-600 text-white font-bold py-3 rounded-xl flex items-center justify-center gap-2 hover:bg-emerald-700 shadow-md text-xs sm:text-sm transition active:scale-95">
                    <Smartphone size={16} /> Kirim Pesanan ke WA
                  </button>
                </div>
              </div>
            );
          })
        )}
      </div>
    </div>
  );

  const renderStore = () => {
    if (!user) {
      if (isAuthChecking) return <div className="min-h-screen flex items-center justify-center bg-gray-50 animate-pulse"><Store size={40} className="text-emerald-300"/></div>;
      
      return (
        <div className="pb-24 bg-gray-50 min-h-screen flex flex-col justify-center px-4">
          <div className="bg-white rounded-3xl shadow-xl border border-gray-100 p-6 max-w-md mx-auto w-full">
            <div className="text-center mb-6">
              <div className="bg-emerald-100 w-14 h-14 rounded-full flex items-center justify-center mx-auto mb-3 text-emerald-600 shadow-inner">
                <Lock size={28} />
              </div>
              <h2 className="text-xl font-black text-gray-800">{authMode === 'login' ? 'Masuk ke Toko' : 'Buka Toko Baru'}</h2>
              <p className="text-xs text-gray-500 mt-1">Kelola jualan Anda di pasar digital syariah.</p>
            </div>
            
            {authError && <div className="bg-red-50 text-red-600 p-3 rounded-xl text-xs mb-4 border border-red-100 flex items-start gap-2"><AlertCircle size={14} className="shrink-0 mt-0.5"/> {authError}</div>}

            <form onSubmit={authMode === 'login' ? handleLogin : handleRegister} className="space-y-3.5">
              <div>
                <input required type="email" placeholder="Email Anda" value={authForm.email} onChange={e => setAuthForm({...authForm, email: e.target.value})} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-4 py-3 focus:ring-2 focus:ring-emerald-500 outline-none text-xs sm:text-sm font-medium" />
              </div>
              <div>
                <input required type="password" placeholder="Password (Min. 6 Karakter)" value={authForm.password} onChange={e => setAuthForm({...authForm, password: e.target.value})} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-4 py-3 focus:ring-2 focus:ring-emerald-500 outline-none text-xs sm:text-sm font-medium" minLength={6} />
              </div>
              
              {authMode === 'register' && (
                <div className="space-y-3.5 pt-3 border-t border-gray-100 mt-3 animate-in fade-in slide-in-from-top-2">
                  <input required type="text" placeholder="Nama Toko (Cth: Warung Bu Siti)" value={authForm.storeName} onChange={e => setAuthForm({...authForm, storeName: e.target.value})} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-4 py-3 focus:ring-2 focus:ring-emerald-500 outline-none text-xs sm:text-sm font-medium" />
                  <input required type="number" placeholder="No. WhatsApp (Cth: 62812...)" value={authForm.waNumber} onChange={e => setAuthForm({...authForm, waNumber: e.target.value})} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-4 py-3 focus:ring-2 focus:ring-emerald-500 outline-none text-xs sm:text-sm font-medium" />
                  <input required type="text" placeholder="Lokasi (Cth: Jakarta Selatan)" value={authForm.location} onChange={e => setAuthForm({...authForm, location: e.target.value})} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-4 py-3 focus:ring-2 focus:ring-emerald-500 outline-none text-xs sm:text-sm font-medium" />
                </div>
              )}
              <button type="submit" disabled={isSubmitting} className="w-full bg-emerald-600 text-white font-black py-3 rounded-xl shadow-lg shadow-emerald-200 hover:bg-emerald-700 transition disabled:opacity-50 mt-1 text-xs sm:text-sm">
                {isSubmitting ? 'Memproses...' : (authMode === 'login' ? 'Masuk' : 'Daftar & Buka Toko')}
              </button>
            </form>
            <div className="mt-5 text-center">
              <button type="button" onClick={() => setAuthMode(authMode === 'login' ? 'register' : 'login')} className="text-gray-500 text-xs font-medium hover:text-emerald-600">
                {authMode === 'login' ? 'Belum punya toko? ' : 'Sudah punya toko? '}
                <span className="text-emerald-600 font-bold underline underline-offset-2">{authMode === 'login' ? 'Daftar di sini' : 'Masuk di sini'}</span>
              </button>
            </div>
          </div>
        </div>
      );
    }

    const sellerProducts = products.filter(p => p.sellerId === user?.uid);

    return (
      <div className="pb-24 bg-gray-50 min-h-screen">
        <div className="bg-emerald-600 px-4 pt-6 pb-12 text-white rounded-b-[35px] shadow-md">
          <div className="flex justify-between items-center mb-5">
            <h2 className="text-lg font-black">Dashboard Toko</h2>
            <div className="flex gap-2">
              <button onClick={() => { setProfileForm({ storeName: storeProfile.storeName, waNumber: storeProfile.waNumber, location: storeProfile.location, storeImage: storeProfile.storeImage || '' }); setIsEditingProfile(true); }} className="bg-white/20 p-2 rounded-full hover:bg-white/30 transition backdrop-blur-sm flex items-center justify-center"><Edit2 size={16}/></button>
              <button onClick={handleLogout} className="bg-white/20 p-2 rounded-full hover:bg-white/30 transition backdrop-blur-sm flex items-center justify-center"><LogOut size={16}/></button>
            </div>
          </div>
          {storeProfile && (
            <div className="flex items-center gap-3.5">
              <div className="w-14 h-14 bg-white rounded-full flex items-center justify-center text-emerald-600 text-xl font-black shadow-lg overflow-hidden border-2 border-white shrink-0">
                {storeProfile.storeImage ? (
                  <img src={storeProfile.storeImage} alt="Logo Toko" className="w-full h-full object-cover" />
                ) : (
                  storeProfile.storeName.charAt(0).toUpperCase()
                )}
              </div>
              <div className="min-w-0">
                <h3 className="font-black text-base sm:text-lg flex items-center gap-1.5 truncate">{storeProfile.storeName} <CheckCircle size={16} className="text-blue-300 fill-white shrink-0"/></h3>
                <p className="text-emerald-100 text-xs flex items-center gap-1 mt-0.5 truncate"><Smartphone size={12} className="shrink-0"/> +{storeProfile.waNumber}</p>
              </div>
            </div>
          )}
        </div>

        <div className="px-4 -mt-5">
          <button onClick={() => { if(isFormOpen) resetForm(); else setIsFormOpen(true); }} className="w-full bg-white text-emerald-600 border-2 border-emerald-500 font-black py-3 rounded-2xl shadow-lg hover:bg-emerald-50 transition flex justify-center items-center gap-2 text-xs sm:text-sm">
            {isFormOpen ? 'Batal Tambah' : <><Plus size={18} /> Tambah Dagangan Baru</>}
          </button>
        </div>

        {isFormOpen && (
          <div className="px-4 mt-4 animate-in slide-in-from-top-4">
            <div className="bg-white rounded-3xl shadow-md border border-gray-100 overflow-hidden">
              <div className="bg-gray-50 px-4 py-3.5 border-b border-gray-100">
                <h3 className="font-black text-gray-800 flex items-center gap-2 text-xs sm:text-sm">
                  {editingProductId ? <Edit2 size={16} className="text-emerald-500"/> : <Store size={16} className="text-emerald-500"/>} 
                  {editingProductId ? 'Edit Barang' : 'Informasi Barang Baru'}
                </h3>
              </div>
              <form onSubmit={handleSubmitProduct} className="p-4 space-y-3.5">
                <div>
                  <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1">Nama Barang</label>
                  <input required type="text" value={newProduct.name} onChange={e => setNewProduct({...newProduct, name: e.target.value})} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-3 py-2.5 focus:ring-2 focus:ring-emerald-500 outline-none text-xs sm:text-sm font-medium" placeholder="Contoh: Nasi Goreng" disabled={isSubmitting} />
                </div>
                <div>
                  <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1">Harga Barang (Rp)</label>
                  <input required type="number" value={newProduct.price} onChange={e => setNewProduct({...newProduct, price: e.target.value})} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-3 py-2.5 focus:ring-2 focus:ring-emerald-500 outline-none text-xs sm:text-sm font-medium" placeholder="Contoh: 50000" disabled={isSubmitting} min="0" />
                </div>
                <div>
                  <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1 flex items-center gap-1"><ImageIcon size={12}/> Foto Barang</label>
                  <div className="bg-gray-50 p-3 rounded-xl border-2 border-gray-200 border-dashed text-center relative hover:bg-gray-100 transition">
                    <input type="file" accept="image/*" onChange={handleImageUpload} className="absolute inset-0 w-full h-full opacity-0 cursor-pointer" disabled={isSubmitting} />
                    {newProduct.imageUrl ? (
                      <div className="relative inline-block">
                        <img src={newProduct.imageUrl} alt="Preview" className="h-28 w-28 object-cover rounded-xl shadow-sm border border-emerald-200" />
                        <span className="absolute -top-2 -right-2 bg-red-500 text-white rounded-full p-1 shadow-md cursor-pointer"><X size={10}/></span>
                      </div>
                    ) : (
                      <div className="text-gray-400 flex flex-col items-center py-2">
                        <Plus size={24} className="mb-1 opacity-50"/>
                        <span className="text-[10px] font-bold">Tap untuk Pilih Foto</span>
                      </div>
                    )}
                  </div>
                </div>
                
                <div className="grid grid-cols-2 gap-2.5">
                  <div>
                    <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1">Kategori</label>
                    <select value={newProduct.category} onChange={e => {
                      const newCat = e.target.value;
                      setNewProduct({...newProduct, category: newCat, subcategory: categoryHierarchy[newCat][0]});
                    }} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-2.5 py-2.5 focus:ring-2 focus:ring-emerald-500 outline-none text-xs font-medium truncate" disabled={isSubmitting}>
                      {Object.keys(categoryHierarchy).map(cat => <option key={cat} value={cat}>{cat}</option>)}
                    </select>
                  </div>
                  <div>
                    <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1">Subkategori</label>
                    <select value={newProduct.subcategory} onChange={e => setNewProduct({...newProduct, subcategory: e.target.value})} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-2.5 py-2.5 focus:ring-2 focus:ring-emerald-500 outline-none text-xs font-medium truncate" disabled={isSubmitting}>
                      {categoryHierarchy[newProduct.category]?.map(sub => <option key={sub} value={sub}>{sub}</option>)}
                    </select>
                  </div>
                </div>
                <div>
                  <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1">Deskripsi Singkat</label>
                  <textarea rows="2" value={newProduct.description} onChange={e => setNewProduct({...newProduct, description: e.target.value})} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-3 py-2.5 focus:ring-2 focus:ring-emerald-500 outline-none text-xs sm:text-sm font-medium" placeholder="Jelaskan spesifikasi..." disabled={isSubmitting}></textarea>
                </div>
                <div>
                  <button type="submit" disabled={isSubmitting} className="w-full bg-emerald-600 text-white font-black py-3 rounded-xl shadow-lg hover:bg-emerald-700 transition disabled:opacity-50 text-xs sm:text-sm">
                    {isSubmitting ? 'Menyimpan...' : (editingProductId ? 'Simpan Perubahan' : 'Tayangkan di Pasar')}
                  </button>
                </div>
              </form>
            </div>
          </div>
        )}

        <div className="px-4 mt-6">
          <h3 className="font-black text-gray-800 text-sm sm:text-base mb-3 flex items-center gap-2">
            <LayoutGrid size={18} className="text-emerald-500"/> Etalase Anda
          </h3>
          {sellerProducts.length === 0 ? (
             <div className="text-center py-10 bg-white rounded-2xl border-2 border-dashed border-gray-200"><p className="text-gray-400 font-bold text-xs">Etalase masih kosong.</p></div>
          ) : (
            <div className="space-y-3">
              {sellerProducts.map(product => (
                <div key={product.id} className="bg-white rounded-2xl shadow-sm border border-gray-100 overflow-hidden flex p-2.5 gap-3 items-center">
                  <img src={product.image} alt={product.name} className="w-20 h-20 object-cover rounded-xl bg-gray-50 shrink-0" />
                  <div className="flex-1 min-w-0">
                    <h4 className="font-bold text-gray-800 text-xs sm:text-sm line-clamp-1 leading-tight mb-0.5">{product.name}</h4>
                    <p className="font-black text-emerald-600 text-xs sm:text-sm mb-2">{formatRupiah(product.price)}</p>
                    
                    {confirmDeleteId === product.id ? (
                      <div className="flex gap-1.5">
                        <button onClick={() => executeDelete(product.id)} className="flex-1 bg-red-500 text-white py-1.5 rounded-lg text-[10px] font-black">Ya, Hapus</button>
                        <button onClick={() => setConfirmDeleteId(null)} className="flex-1 bg-gray-200 text-gray-700 py-1.5 rounded-lg text-[10px] font-bold">Batal</button>
                      </div>
                    ) : (
                      <div className="flex gap-1.5">
                        <button onClick={() => handleEditClick(product)} className="flex-1 border border-emerald-500 text-emerald-600 py-1.5 rounded-lg text-[10px] font-bold flex justify-center items-center gap-1"><Edit2 size={10}/> Edit</button>
                        <button onClick={() => setConfirmDeleteId(product.id)} className="flex-1 border border-red-200 text-red-500 py-1.5 rounded-lg text-[10px] font-bold flex justify-center items-center gap-1"><Trash2 size={10}/> Hapus</button>
                      </div>
                    )}
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>

        {isEditingProfile && (
          <div className="fixed inset-0 bg-black/60 z-[100] flex items-center justify-center p-4 backdrop-blur-sm animate-in fade-in">
            <div className="bg-white rounded-3xl p-5 w-full max-w-sm shadow-2xl max-h-[90vh] overflow-y-auto">
              <h3 className="font-black text-lg text-gray-800 mb-3 flex items-center gap-2">
                <Store className="text-emerald-600"/> Edit Profil Toko
              </h3>
              <form onSubmit={handleSaveProfile} className="space-y-3.5">
                <div>
                  <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1 flex items-center gap-1"><ImageIcon size={12}/> Logo / Foto Toko</label>
                  <div className="bg-gray-50 p-2.5 rounded-xl border-2 border-gray-200 border-dashed text-center relative hover:bg-gray-100 transition">
                    <input type="file" accept="image/*" onChange={handleProfileImageUpload} className="absolute inset-0 w-full h-full opacity-0 cursor-pointer" disabled={isSubmitting} />
                    {profileForm.storeImage ? (
                      <div className="relative inline-block mt-1">
                        <img src={profileForm.storeImage} alt="Preview Logo" className="h-16 w-16 object-cover rounded-full shadow-sm border border-emerald-200" />
                        <span className="absolute -top-1 -right-1 bg-red-500 text-white rounded-full p-1 shadow-md cursor-pointer z-10" onClick={(e) => { e.preventDefault(); setProfileForm({...profileForm, storeImage: ''}); }}><X size={10}/></span>
                      </div>
                    ) : (
                      <div className="text-gray-400 flex flex-col items-center py-1">
                        <Plus size={20} className="mb-1 opacity-50"/>
                        <span className="text-[10px] font-bold">Pilih Foto (Opsional)</span>
                      </div>
                    )}
                  </div>
                </div>

                <div>
                  <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1">Nama Toko Baru</label>
                  <input required type="text" value={profileForm.storeName} onChange={e => setProfileForm({...profileForm, storeName: e.target.value})} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-3 py-2.5 focus:ring-2 focus:ring-emerald-500 outline-none text-xs sm:text-sm font-medium" />
                </div>
                <div>
                  <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1">No. WhatsApp</label>
                  <input required type="number" value={profileForm.waNumber} onChange={e => setProfileForm({...profileForm, waNumber: e.target.value})} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-3 py-2.5 focus:ring-2 focus:ring-emerald-500 outline-none text-xs sm:text-sm font-medium" />
                </div>
                <div>
                  <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1">Lokasi / Alamat</label>
                  <input required type="text" value={profileForm.location} onChange={e => setProfileForm({...profileForm, location: e.target.value})} className="w-full bg-gray-50 border border-gray-200 rounded-xl px-3 py-2.5 focus:ring-2 focus:ring-emerald-500 outline-none text-xs sm:text-sm font-medium" />
                </div>
                <div className="flex gap-2.5 pt-3 border-t border-gray-100">
                  <button type="button" onClick={() => setIsEditingProfile(false)} disabled={isSubmitting} className="flex-1 bg-gray-100 text-gray-600 font-bold py-2.5 rounded-xl hover:bg-gray-200 transition text-xs">Batal styling</button>
                  <button type="submit" disabled={isSubmitting} className="flex-1 bg-emerald-600 text-white font-bold py-2.5 rounded-xl hover:bg-emerald-700 transition shadow-lg shadow-emerald-200 text-xs">
                    {isSubmitting ? 'Menyimpan...' : 'Simpan'}
                  </button>
                </div>
              </form>
            </div>
          </div>
        )}
      </div>
    );
  };

  return (
    <div className="max-w-md mx-auto bg-gray-50 min-h-screen relative shadow-2xl overflow-hidden font-sans">
      {activeTab === 'home' && renderHome()}
      {activeTab === 'cart' && renderCart()}
      {activeTab === 'store' && renderStore()}
      
      <BottomNav />
      <ToastNotification />
    </div>
  );
}
